result <- lm(df$Diabetes ~ . , data = df) # ~.은 모든변수사용 (하나하+하기귀찮아서) 
result1 <- lm(df$Diabetes ~ 1, data = df) # 다른 변수 고려x (단순 평균으로 예측)

anova_result <- anova(result1,result)
