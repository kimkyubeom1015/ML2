library(tidyverse)
library(tidymodels)
library(caret)

df<- read.csv("C:/Users/USER/Downloads/telecom_churn.csv")

skim(df)

df$Churn <- as.factor(df$Churn)  

#데이터 나누기 학습 80%, 테스트 20%
trainIndex <- createDataPartition(df$Churn, p = 0.8, list = FALSE)
train_data <- df[trainIndex,]
test_data <- df[-trainIndex,]


cost_value <- 1 # Cost(C)
gamma_value <- 0.1 # Gamma

# SVM모델 생성
svm_model <- svm(train_data$Churn ~ ., data = train_data,
                 type = 'C-classification', #Soft margin
                 kernel = 'polynomial',     #다항커널
                 cost = cost_value,
                 gamma = gamma_value,    #확률값으로 도출
                probability = TRUE)

summary(svm_model)


library(pROC)
predicted_probs <- predict(svm_model, test_data, probability = TRUE)
predicted_probs   #yes일 확률,no일 확률 둘이 더하면 1 

predicted_probs <- attr(predicted_probs, "probabilities")[, "Yes"] #객체의 속성을 추출하는 함수
predicted_probs

predicted_classes <- ifelse(predicted_probs > 0.5, "Yes", "No")
predicted_classes <- as.factor(predicted_classes)
predicted_classes

conf_matrix <- confusionMatrix(predicted_classes, test_data$Churn)
print(conf_matrix) #ACU : 0.8745  정확도 : 0.84

df <- read.csv("C:/Users/USER/Desktop/Final_Augmented_dataset_Diseases_and_Symptoms.csv", header = TRUE , na =".")

#80퍼센트는 트레인데이터, 20퍼센트는 테스트데이터로 쓰겠다고 지정

train_data <- df[trainIndex,]

#아노바 분석으로 모델 전체가 유의한가를 검사
result1=glm(train_data$anxiety.and.nervousness ~ 1, data = train_data, family ="binomial")
result2=glm(train_data$anxiety.and.nervousness ~ ., data = train_data, family ="binomial")

anova(result1,result2,test="Chisq")

summary(result2)

library(car)

#Kappa(데이터가 한쪽이 너무 많을 때, 한쪽의 값을 도출해도 정확도가 높기 때문에 우연으로 맞췄는지 실제로 맞췄는지를

#Kappa값이 1에 가까울 수록 좋은 모델, -값을 가지면 무작위로 예측하는 것 보다 정확도가 낮다는 의미(과적합 판단)
